fork() — Como se bifurca un proceso

2.0s
Pulsa "Siguiente" o "Auto" para comenzar.

procesos.c

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>

int main( ) {
  int contador = 0;
  pid_t pid;

  printf( "Proceso inicial (%d): contador = %d\n", getpid( ), contador );

  pid = fork( );

  if ( pid != 0 ) {
    /* PADRE */
    contador = contador + 10;
    printf( "Padre (%d): contador = %d\n", getpid( ), contador );
    printf( "Padre (%d): esperando al hijo...\n", getpid( ) );
    waitpid( pid, NULL, 0 );
    printf( "Padre (%d): el hijo termino. Fin.\n", getpid( ) );
  } else {
    /* HIJO */
    contador = contador + 1;
    printf( "Hijo (%d): contador = %d\n", getpid( ), contador );
    printf( "Hijo (%d): haciendo mi trabajo...\n", getpid( ) );
    contador = contador + 1;
    printf( "Hijo (%d): contador = %d. Fin.\n", getpid( ), contador );
  }

  exit( 0 );
}
